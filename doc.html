1. System Overview

SMART GRADE is a Progressive Web Application (PWA) for grade management, designed for SIN GBHS FOUMBAN Form 5B Science. The application works 100% offline after first installation and stores all data locally on the user's device.

Key Features: 14 subjects, 6 sequences, 3 terms, 23 badges, 20 themes, 12 fonts, biometric login, QR code transfer, cloud backup, PWA offline support.
Data Privacy: All data stays on the user's device. No external servers store personal data unless the user explicitly uses cloud backup.

2. Technology Stack

HTML5

Semantic markup, localStorage API, Canvas API, MediaDevices API, Notification API, Vibration API, Clipboard API

CSS3

Flexbox, Grid Layout, CSS Animations, CSS Variables (Custom Properties), Glassmorphism (backdrop-filter), Media Queries, 20 Color Themes

GitHub Gist API (cloud transfer), QR Code generation, Chart.js for graphs, Google Fonts, Font Awesome icons

3. Application Architecture

The application follows a modular architecture with separate JavaScript files for different concerns:

  • utils.js - Core utility functions (formatting, calculations, export/import)
  • database.js - All localStorage operations (CRUD for students, grades, subjects)
  • auth.js - PIN and fingerprint authentication
  • app.js - UI initialization (themes, menu, particles, notifications)
  • transfer.js - Cloud and QR code data transfer
  • confirm-dialog.js - Custom confirmation dialogs
  • install-handler.js - PWA installation across browsers
  • cloud-sync.js - Automatic cloud backup
  • ai-assistant-core.js - AI chat and model management
  • ai-assistant-ui.js - Mermaid diagrams and zoom

JavaScript 

// ============================================
// SUBJECT MANAGEMENT

/**
 * Returns the subjects selected by a student for a term
 * @param {number} id - Student ID
 * @param {number} term - Term number (1-3)
 * @returns {Array} Array of subject IDs
 */
function getStudentSelectedSubjects(id, term) {
  try {
    var data = localStorage.getItem('smartgrade_selected_' + id + '_term' + term);
    return data ? JSON.parse(data) : DEFAULT_SUBJECTS.map(function(s) { return s.id; });
  } catch(e) {
    return DEFAULT_SUBJECTS.map(function(s) { return s.id; });
  }
}

/**
 * Saves selected subjects for a student and term
 * @param {number} id - Student ID
 * @param {number} term - Term number (1-3)
 * @param {Array} subjects - Array of subject IDs
 */
function saveStudentSelectedSubjects(id, term, subjects) {
  localStorage.setItem('smartgrade_selected_' + id + '_term' + term, JSON.stringify(subjects));
}

/**
 * Returns all subject coefficients for a student
 * @param {number} id - Student ID
 * @returns {Object} Object mapping subjectId to coefficient
 */
function getSubjectCoefficients(id) {
  try {
    var data = localStorage.getItem('smartgrade_coeffs_' + id);
    if (!data) {
      var c = {};
      DEFAULT_SUBJECTS.forEach(function(s) { c[s.id] = 5; });
      return c;
    }
    return JSON.parse(data);
  } catch(e) {
    var c = {};
    DEFAULT_SUBJECTS.forEach(function(s) { c[s.id] = 5; });
    return c;
  }
}

/**
 * Returns the coefficient for a specific subject
 * @param {number} id - Student ID
 * @param {number} subjectId - Subject ID (1-14)
 * @returns {number} Coefficient (1-10)
 */
function getSubjectCoefficient(id, subjectId) {
  var coeffs = getSubjectCoefficients(id);
  return coeffs[subjectId] || 5;
}

/**
 * Returns active subjects with coefficients for a term
 * @param {number} id - Student ID
 * @param {number} term - Term number (1-3)
 * @returns {Array} Array of subject objects with coefficient
 */
function getActiveSubjectsWithCoefficients(id, term) {
  var selected = getStudentSelectedSubjects(id, term);
  return DEFAULT_SUBJECTS.filter(function(s) {
    return selected.indexOf(s.id) !== -1;
  }).map(function(s) {
    s.coefficient = getSubjectCoefficient(id, s.id);
    return s;
  });
}

10. Progressive Web App (PWA)

The application is a fully functional PWA that works offline after first installation.

JavaScript 

// ============================================
// SERVICE WORKER (sw.js)
// ============================================

var CACHE_NAME = 'smartgrade-v5';
var STATIC_ASSETS = [
  './',
  './index.html',
  './dashboard.html',
  './css/base.css',
  './css/layout.css',
  './js/app.js',
  './js/database.js',
  // ... all assets
];

self.addEventListener('install', function(event) {
  event.waitUntil(
    caches.open(CACHE_NAME).then(function(cache) {
      return cache.addAll(STATIC_ASSETS);
    })
  );
});

self.addEventListener('activate', function(event) {
  event.waitUntil(
    caches.keys().then(function(keys) {
      return Promise.all(
        keys.map(function(key) {
          if (key !== CACHE_NAME) {
            return caches.delete(key);
          }
        })
      );
    })
  );
});

self.addEventListener('fetch', function(event) {
  event.respondWith(
    caches.match(event.request).then(function(cached) {
      return cached || fetch(event.request);
    })
  );
});

// ============================================
// INSTALL HANDLER (install-handler.js)
// ============================================

// Disable native browser prompt
window.addEventListener('beforeinstallprompt', function(e) {
  e.preventDefault();
  deferredPrompt = e;
  showCustomInstallBanner();
});

// Install app
function installApp() {
  if (deferredPrompt) {
    deferredPrompt.prompt();
    deferredPrompt.userChoice.then(function(result) {
      if (result.outcome === 'accepted') {
        localStorage.setItem('smartgrade_installed', 'true');
      }
      deferredPrompt = null;
    });
  }
}
The Service Worker caches all HTML, CSS, JS, and assets. The app works 100% offline after the first successful load.

11. Cloud Transfer System

The cloud transfer system uses GitHub Gist API (free, unlimited) to store temporary transfer data.

JavaScript 

// ============================================
// TRANSFER SYSTEM (transfer.js)
// ============================================

/**
 * Generate a transfer code
 * @param {number} studentId - Student ID
 * @returns {string} 6-character transfer code
 */
function generateTransferCode(studentId) {
  var chars = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789';
  var code = '';
  for (var i = 0; i < 6; i++) {
    code += chars.charAt(Math.floor(Math.random() * chars.length));
  }
  return code;
}

/**
 * Store transfer data
 * @param {string} code - Transfer code
 * @param {number} studentId - Student ID
 * @param {string} data - JSON string of user data
 */
function storeTransferData(code, studentId, data) {
  var transferData = {
    studentId: studentId,
    data: data,
    expires: Date.now() + 10 * 60 * 1000 // 10 minutes
  };
  localStorage.setItem('smartgrade_transfer_' + code, JSON.stringify(transferData));
}

/**
 * Get transfer data
 * @param {string} code - Transfer code
 * @returns {Object|null} Transfer data or null if expired/invalid
 */
function getTransferData(code) {
  var data = localStorage.getItem('smartgrade_transfer_' + code);
  if (!data) return null;
  
  var parsed = JSON.parse(data);
  if (parsed.expires < Date.now()) {
    localStorage.removeItem('smartgrade_transfer_' + code);
    return null;
  }
  return parsed;
}

/**
 * Generate QR code
 * @param {string} code - Transfer code
 * @param {string} containerId - DOM element ID for QR code
 */
function generateQRCode(code, containerId) {
  // Uses QRCode.js library
  new QRCode(document.getElementById(containerId), {
    text: window.location.origin + '/transfer.html?code=' + code,
    width: 200,
    height: 200,
    colorDark: '#0f3b48',
    colorLight: '#ffffff',
    correctLevel: QRCode.CorrectLevel.H
  });
}
Important: Transfer codes expire after 10 minutes for security. QR codes are also valid for 10 minutes.

  };

}

/**
 * Import data from JSON backup
 * @param {number} studentId - Student ID
 * @param {string} jsonData - JSON string of user data
 * @returns {Object} { success: boolean, message: string }
 */
function importCompleteUserData(studentId, jsonData) {
  try {
    var data = typeof jsonData === 'string' ? JSON.parse(jsonData) : jsonData;
    
    // Restore all data
    if (data.grades) localStorage.setItem('smartgrade_grades_' + studentId, JSON.stringify(data.grades));
    if (data.subjects) {
      if (data.subjects.term1) localStorage.setItem('smartgrade_selected_' + studentId + '_term1', JSON.stringify(data.subjects.term1));
      if (data.subjects.term2) localStorage.setItem('smartgrade_selected_' + studentId + '_term2', JSON.stringify(data.subjects.term2));
      if (data.subjects.term3) localStorage.setItem('smartgrade_selected_' + studentId + '_term3', JSON.stringify(data.subjects.term3));
    }
    if (data.coeffs) localStorage.setItem('smartgrade_coeffs_' + studentId, JSON.stringify(data.coeffs));
    if (data.achievements) localStorage.setItem('smartgrade_achievements_' + studentId, JSON.stringify(data.achievements));
    if (data.goal !== undefined) localStorage.setItem('smartgrade_goal_' + studentId, data.goal);
    if (data.streak) localStorage.setItem('smartgrade_streak_' + studentId, JSON.stringify(data.streak));
    if (data.profile) localStorage.setItem('smartgrade_profile_' + studentId, JSON.stringify(data.profile));
    if (data.history) localStorage.setItem('smartgrade_history_' + studentId, JSON.stringify(data.history));
    
    return { success: true, message: 'Data imported successfully' };
  } catch(e) {
    return { success: false, message: 'Invalid data: ' + e.message };
  }
}
The JSON backup includes all user data: grades, subjects, coefficients, achievements, streak, profile, flashcards, and history.

15. Homework Manager

The Homework Manager allows teachers to create and manage assignments, and students to view them.

Features

  • Add Homework: Teachers can create assignments with title, subject, due date, and description
  • File Attachments: Support for images, videos, audio, PDFs, Word documents, and text files
  • Cloud Sync: All homework data is stored and synced via GitHub Gist
  • Filter by Subject: Students can filter homework by subject
  • Edit/Delete: Teachers can edit or delete assignments
JavaScript 

// ============================================
// HOMEWORK MANAGER (admin-homework.js)
// ============================================

// GitHub Gist configuration
var GITHUB_TOKEN = 'your_github_token';
var GITHUB_GIST_ID = 'your_gist_id';

// Load homework from GitHub
async function loadHomeworks() {
  var response = await fetch('https://api.github.com/gists/' + GITHUB_GIST_ID, {
    headers: { 'Authorization': 'token ' + GITHUB_TOKEN }
  });
  var gist = await response.json();
  return JSON.parse(gist.files['homeworks.json'].content);
}

// Save homework to GitHub
async function saveHomeworks(data) {
  await fetch('https://api.github.com/gists/' + GITHUB_GIST_ID, {
    method: 'PATCH',
    headers: {
      'Authorization': 'token ' + GITHUB_TOKEN,
      'Content-Type': 'application/json'
    },
    body: JSON.stringify({
      files: { 'homeworks.json': { content: JSON.stringify(data) } }
    })
  });
}
Note: The Homework Manager uses GitHub Gist for cloud storage. All data is synced automatically.
